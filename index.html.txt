// ── Rule Engine (client-side demo)
function makeAlert(level, title, body) {
  return `<div class="alert alert-${level}"><div class="alert-icon">${level==='red'?'⚠️':level==='yellow'?'▲':'✅'}</div><div><div class="alert-title">${title}</div><div class="alert-body">${body}</div></div></div>`;
}
function checkBP() {
  var sys = parseInt(document.getElementById('sys').value) || 0;
  var dia = parseInt(document.getElementById('dia').value) || 0;
  var h = document.getElementById('bp-hint');
  if (!sys && !dia) { h.innerHTML = ''; return; }
  if (sys >= 180 || dia >= 120)
    h.innerHTML = makeAlert('red','高血壓危象風險','收縮壓 ≥180 或舒張壓 ≥120，請立即休息並聯繫醫師或撥打 119。');
  else if (sys >= 140 || dia >= 90)
    h.innerHTML = makeAlert('yellow','血壓偏高','超過目標值 140/90 mmHg。確認今日降壓藥是否已服用，避免高鈉飲食。');
  else
    h.innerHTML = makeAlert('green','血壓良好','數值在目標範圍（&lt;140/90）內，繼續維持。');
}
function checkBG() {
  var pp = parseInt(document.getElementById('pp').value) || 0;
  var h = document.getElementById('bg-hint');
  if (!pp) { h.innerHTML = ''; return; }
  if (pp > 200) h.innerHTML = makeAlert('red','高血糖警告','餐後血糖 >200，建議今日減少精緻澱粉，並聯繫醫師。');
  else if (pp > 180) h.innerHTML = makeAlert('yellow','餐後血糖偏高','建議飯後散步 15 分鐘，明天早餐以糙米替代白飯，減少 1/3 份量。');
  else h.innerHTML = makeAlert('green','餐後血糖良好','數值在目標範圍 (&lt;180) 內，繼續維持飲食控制。');
}
function checkKidney() {
  var egfr = parseInt(document.getElementById('egfr').value) || 0;
  var h = document.getElementById('kidney-hint');
  if (!egfr) { h.innerHTML = ''; return; }
  if (egfr < 15) h.innerHTML = makeAlert('red','CKD G5 — 腎衰竭，請立即就醫','eGFR &lt;15，需緊急評估腎臟替代療法（透析/移植）。');
  else if (egfr < 30) h.innerHTML = makeAlert('red','CKD G4 — 請儘速回診','eGFR &lt;30，建議立即回診腎臟科，嚴格限蛋白質 0.6g/kg/日。');
  else if (egfr < 45) h.innerHTML = makeAlert('yellow','CKD G3b — 需密切追蹤','蛋白質限制 0.6–0.8g/kg/日，避免 NSAIDs，每3個月回診一次。');
  else if (egfr < 60) h.innerHTML = makeAlert('yellow','CKD G3a','腎功能中度下降，嚴格控制血壓與血糖，每6個月追蹤。');
  else h.innerHTML = makeAlert('green','腎功能尚可','繼續維持現有治療方針，每年定期追蹤。');
}

// ── Save
function saveRecord() {
  var btn = document.querySelector('.save-btn');
  var msg = document.getElementById('save-success');
  btn.querySelector('span').textContent = '儲存中...';
  btn.disabled = true;
  setTimeout(function() {
    btn.querySelector('span').textContent = '確認儲存';
    btn.disabled = false;
    msg.style.display = 'block';
    setTimeout(function() { msg.style.display = 'none'; }, 3000);
  }, 700);
}

// ── Tasks
var taskDone = 2, taskTotal = 5;
function toggleTask(item) {
  var cb = item.querySelector('.task-cb');
  var txt = item.querySelector('.task-text');
  var wasDone = cb.classList.contains('done');
  cb.classList.toggle('done');
  txt.classList.toggle('done');
  cb.textContent = cb.classList.contains('done') ? '✓' : '';
  taskDone += wasDone ? -1 : 1;
  var pct = Math.round(taskDone / taskTotal * 100);
  document.getElementById('task-prog').style.width = pct + '%';
  document.getElementById('task-count-lbl').textContent = taskDone + '/' + taskTotal + ' 完成';
  document.getElementById('home-prog').style.width = pct + '%';
  document.getElementById('home-task-count').textContent = taskDone + '/' + taskTotal + ' 完成';
}
